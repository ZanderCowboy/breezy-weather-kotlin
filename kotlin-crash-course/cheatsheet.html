Coroutines

// Async programming
suspend fun loadData(): Data {
    return withContext(Dispatchers.IO) {
        api.fetchData()
    }
}

// Launch coroutines
viewModelScope.launch {
    val data = loadData()
    updateUI(data)
}

// Parallel execution
val user = async { getUser() }
val posts = async { getPosts() }
combine(user.await(), posts.await())

Flow

// Reactive streams
fun observeData(): Flow<Data> = flow {
    while (true) {
        emit(fetchLatest())
        delay(1000)
    }
}

// Transform data
val names = users
    .map { it.name }
    .filter { it.isNotEmpty() }
    .distinctUntilChanged()

// Collect in UI
data.collect { updateUI(it) }

Sealed Classes

// Type-safe state management
sealed class Result<T> {
    data class Success<T>(val data: T) : Result<T>()
    data class Error<T>(val message: String) : Result<T>()
    class Loading<T> : Result<T>()
}

// Exhaustive when
when (result) {
    is Result.Success -> showData(result.data)
    is Result.Error -> showError(result.message)
    is Result.Loading -> showLoading()
}

Null Safety

// Nullable types
var name: String? = null

// Safe calls
name?.length
name?.uppercase()

// Elvis operator
val length = name?.length ?: 0

// let for non-null
name?.let { println(it) }

// Not-null assertion
name!!.length // Throws if null

When Expressions

// Smart switch
when (status) {
    Status.LOADING -> showSpinner()
    Status.SUCCESS -> hideSpinner()
    else -> showError()
}

// Expression form
val message = when (code) {
    200 -> "OK"
    404 -> "Not Found"
    in 500..599 -> "Server Error"
    else -> "Unknown"
}

Higher-Order Functions

// Functions as parameters
fun processData(
    data: List<String>,
    transform: (String) -> String
): List<String> = data.map(transform)

// Usage
processData(names) { it.uppercase() }

// Function types
typealias Validator = (String) -> Boolean
val emailValidator: Validator = { it.contains("@") }

Scope Functions

// apply - configure & return object
val user = User().apply {
    name = "Alice"
    age = 25
}

// let - transform non-null
user?.let { saveToDatabase(it) }

// run - execute block
val result = user.run {
    "Name: $name, Age: $age"
}

// also - side effects
user.also { println("User: $it") }
